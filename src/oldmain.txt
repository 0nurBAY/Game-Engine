    #version 330 core

    layout (location = 0) in vec3 aPos;
    layout (location = 1) in vec2 aTexCoord;

    out vec2 TexCoord;
    uniform vec2 uvMax;
    uniform vec2 uvMin;

    uniform mat4 model;
    uniform mat4 view;
    uniform mat4 projection;
    void main()
    {
        gl_Position = projection * view * model * vec4(aPos, 1.0);
        TexCoord = uvMin + aTexCoord*(uvMax-uvMin);
    }
    )";
    const char* fragmentShaderSource_sprite = R"(
    #version 330 core

    in vec2 TexCoord;
    out vec4 FragColor;

    uniform sampler2D ourTexture;
    
    void main()
    {
        FragColor = texture(ourTexture,TexCoord);
    }
    )";

    #version 330 core

    layout (location = 0) in vec3 aPos;

    uniform mat4 model;
    uniform mat4 view;
    uniform mat4 projection;
    void main()
    {
        gl_Position = projection * view * model * vec4(aPos, 1.0);
    }
    )";

    #version 330 core

    out vec4 FragColor;
    uniform vec4      solidColor;
    void main()
    {
                FragColor = solidColor;
    }
    )";

    #version 330 core

    layout (location = 0) in vec3 aPos;

    out vec2 LocalPos;

    uniform mat4 model;
    uniform mat4 view;
    uniform mat4 projection;
    void main()
    {
        gl_Position = projection * view * model * vec4(aPos, 1.0);
        LocalPos = aPos.xy;
    }
    )";

    #version 330 core

    in vec2 LocalPos;
    out vec4 FragColor;

    uniform vec4      gradientCenter;
    uniform vec4      gradientEdge;
    void main()
    {
        float distance = length(LocalPos)/200.0;
        distance = clamp(distance, 0.0,1.0);
        FragColor = mix(gradientCenter,gradientEdge,distance);
    }
    )";
